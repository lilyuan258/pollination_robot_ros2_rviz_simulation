```bash
cd ~/ros2_ws
source /opt/ros/jazzy/setup.bash
colcon build --packages-select only_robot_arm only_robot_arm_moveit_config --symlink-install
source ~/ros2_ws/install/setup.bash

# 1) 仅显示模型 + 花朵 marker（不跑 MoveIt 循环）
ros2 launch only_robot_arm display_jazzy.launch.py

# 2) MoveIt 授粉循环仿真（推荐）
ros2 launch only_robot_arm_moveit_config demo.launch.py

# 3) 无 RViz 跑闭环（用于日志验证）
ros2 launch only_robot_arm_moveit_config demo.launch.py use_rviz:=false run_cycle:=true

# 4) 调整花朵位置后重复仿真（示例）
ros2 launch only_robot_arm_moveit_config demo.launch.py \
  dx_world:=0.15 dy_world:=-0.65 flower_center_z:=0.12

# 5) 调整“先基座顺时针旋转角”（默认 pi/2）
ros2 launch only_robot_arm_moveit_config demo.launch.py \
  base_clockwise_delta_rad:=1.57079632679
```
